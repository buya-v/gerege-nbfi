### corpus (each line discriminates class-reading from literal-reading)
     1	x1y
     2	xdy
     3	x y
     4	xsy
     5	x_y
     6	xwy

  x1y : matches x\dy ONLY if \d = digit class
  xdy : matches x\dy ONLY if \d = literal 'd'
  x y : matches x\sy ONLY if \s = whitespace class
  xsy : matches x\sy ONLY if \s = literal 's'
  x_y : matches x\wy ONLY if \w = word class
  xwy : matches x\wy ONLY if \w = literal 'w'

### BSD grep -E (/usr/bin/grep) -- reading per escape
  pattern x\dy ->
      1:x1y
  pattern x\sy ->
      3:x y
  pattern x\wy ->
      1:x1y
      2:xdy
      4:xsy
      5:x_y
      6:xwy
