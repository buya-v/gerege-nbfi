// t64cf emits, as JSON, the schedule the Go port produces for each pass-3g
// parity candidate. It is run once on a pristine scratch copy of the port (the
// CONTROL) and once per named change (the COUNTERFACTUALS).
//
// It contains NO observed value and NO expectation. The comparison against what
// the reference oracle emitted is done afterwards, in Python, against the
// committed capture artefact.
//
// Money is int64 minor units. No floating point.
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"

	"github.com/gerege/nexus/internal/apps/loanschedule"
	"github.com/gerege/nexus/internal/apps/loanschedule/contract"
)

type row struct {
	Kind        string `json:"kind"`
	Number      int32  `json:"installment_number"`
	FromDate    string `json:"from_date"`
	DueDate     string `json:"due_date"`
	Principal   int64  `json:"principal_minor"`
	Interest    int64  `json:"interest_minor"`
	Outstanding int64  `json:"outstanding_principal_minor"`
}

type out struct {
	Case string `json:"case"`
	Rows []row  `json:"rows"`
}

func main() {
	cases := []struct {
		id        string
		principal int64
		n         int32
		num, den  int64
	}{
		{"T64-ZP-A", 28, 56, 27, 125},
		{"T64-ZP-B", 28, 55, 27, 125},
		{"T64-ZP-C", 17, 34, 9, 25},
		{"T64-ZP-D", 36, 72, 21, 125},
	}
	start := contract.CivilDate{Year: 2024, Month: 1, Day: 1}
	gen := loanschedule.New()
	var all []out
	for _, c := range cases {
		req := contract.GenerateRequest{
			TimeZone:                  "Asia/Ulaanbaatar",
			Currency:                  contract.Currency{Code: "MNT", MinorUnitDigits: 2},
			Rounding:                  contract.Rounding{SignificantDigits: 19, RateFactorScale: 19, Mode: contract.RoundingHalfUp},
			ScheduleStartDate:         start,
			Disbursements:             []contract.Disbursement{{Date: start, AmountMinor: c.principal}},
			NumberOfRepayments:        c.n,
			RepaymentEvery:            1,
			RepaymentFrequencyUnit:    contract.FrequencyMonths,
			AnnualNominalInterestRate: contract.Rate{Numerator: c.num, Denominator: c.den},
			InterestMethod:            contract.InterestMethodDecliningBalance,
			DayCount:                  contract.DayCountFixed30Over360,
			DownPaymentPercentage:     contract.Rate{Numerator: 0, Denominator: 1},
		}
		s, err := gen.Generate(context.Background(), req)
		if err != nil {
			fmt.Fprintln(os.Stderr, c.id, err)
			os.Exit(1)
		}
		o := out{Case: c.id}
		for _, p := range s.Periods {
			k := "REPAYMENT"
			if p.Kind == contract.PeriodKindDisbursement {
				k = "DISBURSEMENT"
			} else if p.Kind == contract.PeriodKindDownPayment {
				k = "DOWN_PAYMENT"
			}
			o.Rows = append(o.Rows, row{
				Kind:        k,
				Number:      p.InstallmentNumber,
				FromDate:    fmt.Sprintf("%04d-%02d-%02d", p.FromDate.Year, p.FromDate.Month, p.FromDate.Day),
				DueDate:     fmt.Sprintf("%04d-%02d-%02d", p.DueDate.Year, p.DueDate.Month, p.DueDate.Day),
				Principal:   p.PrincipalMinor,
				Interest:    p.InterestMinor,
				Outstanding: p.OutstandingPrincipalMinor,
			})
		}
		all = append(all, o)
	}
	enc := json.NewEncoder(os.Stdout)
	enc.SetIndent("", "  ")
	if err := enc.Encode(all); err != nil {
		panic(err)
	}
}
