#!/usr/bin/env python3
"""T196 RED FIXTURE — a registrar that is imported but NEVER LIVE.

`mylib_guard.install_restore` really does register an atexit restore, but here it is
only ever called from `setup()`, which nothing calls.  At run time no handler is
ever registered, so the DEC-1 truncation below is genuinely unprotected.

T179 already drops a LOCAL registration made inside a function nothing calls
(`process_handlers`: "A registration inside a function nothing calls is NOT live").
T196-3 follows one import hop for handlers, so it must inherit that same
conservatism — otherwise the import hop is strictly more permissive than the local
rule it is supposed to mirror.

REQUIRED: UNGUARDED / TRUSTED, `--enforce` exit 1 — both before and after T196-3.

INERT.  Held as `.py.txt`; every write is inside a function nothing calls.
"""
import io
import mylib_guard

P = "docs/adr/DEC-1-schedule-generator-adapter.md"

if __name__ == "__main__":
    raise SystemExit("t196 fixture: parsed, never run")


def setup():
    mylib_guard.install_restore(P)      # never reached: nothing calls setup()


def never_called():
    s = io.open(P, encoding="utf-8").read()
    io.open(P, "w", encoding="utf-8").write(s.replace("a", "b"))   # T196-SITE
