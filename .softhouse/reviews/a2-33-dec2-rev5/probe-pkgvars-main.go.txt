package main

// A2-33: measure I3-PKG-STATE's population (c.PkgVars) — package-level var NAMES — under a root,
// replicating ledgerguard's own loop (GenDecl, token.VAR, ValueSpec, vs.Names) and prunedDirs.

import (
	"fmt"
	"go/ast"
	"go/parser"
	"go/token"
	"io/fs"
	"os"
	"path/filepath"
	"strings"
)

var prunedDirs = map[string]bool{".git": true, "vendor": true, "node_modules": true}

func main() {
	root := os.Args[1]
	n := 0
	files := 0
	fset := token.NewFileSet()
	err := filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if d.IsDir() {
			if prunedDirs[d.Name()] {
				return filepath.SkipDir
			}
			return nil
		}
		if !strings.HasSuffix(path, ".go") {
			return nil
		}
		files++
		f, e := parser.ParseFile(fset, path, nil, parser.ParseComments)
		if e != nil {
			return e
		}
		for _, dec := range f.Decls {
			gd, ok := dec.(*ast.GenDecl)
			if !ok || gd.Tok != token.VAR {
				continue
			}
			for _, spec := range gd.Specs {
				vs, ok := spec.(*ast.ValueSpec)
				if !ok {
					continue
				}
				n += len(vs.Names)
			}
		}
		return nil
	})
	if err != nil {
		fmt.Println("ERR", err)
		os.Exit(2)
	}
	fmt.Printf("A2-33 I3-PKG-STATE population (package-level var names) = %d across %d files under %s\n", n, files, root)
}
